C++ / Embedded C++

class GPIO{
public:
 constexpr GPIO(uint32_t p,uint8_t pin):port_(p),pin_(pin){}
 void set(){ *reinterpret_cast<volatile uint32_t*>(port_) |= (1U<<pin_); }
private: uint32_t port_; uint8_t pin_;
};

Rules: No exception, No RTTI, No new in hot path, RAII.
